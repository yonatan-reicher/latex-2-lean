                                     Tokens

tokens ::= (\s*token)*\s*
token ::= word | command | number | symbol | error
word ::= \w+
command ::= \\(\w|\d)+
number ::= \d+
symbol ::= [=+-*/(){},;] // and maybe some more
error ::= \S+



                                    Parsers

expr ::= binaryExpr
binaryExpr ::=
  | applicationTerm ( binOp applicationTerm )*

applicationExpr ::= atom+

atom ::= 
  | emptySetCommand
  | range
  | number
  | "\abs" atom
  | "\sum" atom
  | "{" expr "}"
  | "\{" setInsides "\}"
  | "\set" "{" setInsides "}"
  | "\mset" "{" setInsides "}"
  | "(" ( expr "," )* expr ")"
  | quantifiedExpr
  | word // variable name

emptySetCommand ::= "\emptySet" | "\varnothing"

range ::= number ".."  number

setInsides ::=
  | // empty set
  | expr "\mid" ( expr "," )* expr // map
  | ( expr "," )* expr // simple set

quantifiedExpr ::= ("\forall" | "\exists") expr "," expr

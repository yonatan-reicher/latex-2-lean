                                     Tokens

tokens ::= (\s*token)*\s*
token ::= word | command | number | symbol | error
word ::= \w+
command ::= \\(\w|\d)+
number ::= \d+
symbol ::= [=+-*/(){},;] // and maybe some more
error ::= \S+



                                    Parsers

expr ::= equalityExpr

equalityExpr ::= addOrSubExpr ( ("=" | "\neq" | "\leq" | "\geq" | "<" | ">") addOrSubExpr)*

addOrSubExpr ::= termExpr ( ("+" | "-" | "\cup" | "\cap" | "\setminus") termExpr )*

termExpr ::= atom ( ("*" | "/" | \"times" | "\"

binaryExpr ::=
  | atom ( binOp atom )*

atom ::= 
  | "\emptySet"
  | "\varnothing"
  | number ".."  number
  | number
  | "\abs" atom
  | "\sum" atom
  | "{" expr "}"
  | "\{" setInsides "\}"
  | "\set" "{" setInsides "}"
  | "\mset" "{" setInsides "}"
  | "(" ( expr "," )* expr ")"
  | word // variable name
  // TODO: add forall?

setInsides ::=
  | // empty set
  | expr "\mid" ( binderOrCond "," )* binderOrCond // map
  | ( expr "," )* expr // simple set

binder ::=
  | word "\in" expr // TODO: shouldn't this be an atom instead of an expr?
  | "\exists" binder "," binder

binderOrCond ::= binder | expr
